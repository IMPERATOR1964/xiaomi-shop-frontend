export default function CatalogPage() {
  const { category } = useParams();
  const { findBySlug } = useCategories();
  const activeCategory = category || 'all';

  const [allProducts, setAllProducts] = useState([]);
  const [filters, setFilters] = useState({});
  const [sortBy, setSortBy]   = useState('newest');

  // Загрузка товаров категории с сервера
  useEffect(() => {
    productsApi.filter({ categoryId: findBySlug(activeCategory)?.backendId })
      .then(res => setAllProducts(res.items));
  }, [activeCategory]);

  // Опции фильтра строятся из полного набора товаров категории
  const filterOptions = useMemo(() => {
    const conf = FILTER_CONFIG[activeCategory] || [];
    return conf.map(({ key, label }) => ({
      key, label,
      values: [...new Set(allProducts.map(p => p.specs?.[key]).filter(Boolean))],
    }));
  }, [activeCategory, allProducts]);

  return (
    <div className="catalog-page">
      <CatalogFilter options={filterOptions} filters={filters} onChange={setFilters} />
      {/* список карточек товаров с учётом фильтров и сортировки */}
    </div>
  );
}
