export default function ResetPasswordPage() {
  const { resetPassword } = useAuth();
  const navigate = useNavigate();
  const { state } = useLocation();
  const [email] = useState(state?.email || '');
  const [code, setCode] = useState('');
  const [newPassword, setNewPassword] = useState('');
  const [error, setError] = useState('');

  // Установка нового пароля по коду из письма
  const handleSubmit = async (e) => {
    e.preventDefault();
    if (newPassword.length < 8)
      return setError('Пароль должен содержать минимум 8 символов');
    try {
      await resetPassword({ email, code, newPassword });
      navigate('/login');
    } catch (err) {
      setError(err.message || 'Не удалось изменить пароль');
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      {error && <div className="auth-error">{error}</div>}
      <input value={code} onChange={e => setCode(e.target.value)} placeholder="Код из письма" />
      <input type="password" value={newPassword}
             onChange={e => setNewPassword(e.target.value)} placeholder="Новый пароль" />
      <button type="submit" className="auth-submit">Сохранить пароль</button>
    </form>
  );
}
