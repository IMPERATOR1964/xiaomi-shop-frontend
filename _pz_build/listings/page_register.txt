export default function RegisterPage() {
  const [username, setUsername]   = useState('');
  const [email, setEmail]         = useState('');
  const [password, setPassword]   = useState('');
  const [password2, setPassword2] = useState('');
  const [error, setError]         = useState('');
  const { register } = useAuth();
  const navigate = useNavigate();

  const handleSubmit = async (e) => {
    e.preventDefault();
    setError('');
    // Проверка корректности введённых данных
    if (!username || !email || !password || !password2)
      return setError('Заполните все поля');
    if (username.length < 3 || username.length > 20)
      return setError('Логин должен быть от 3 до 20 символов');
    if (password !== password2) return setError('Пароли не совпадают');
    if (password.length < 8)
      return setError('Пароль должен содержать минимум 8 символов');

    try {
      const targetEmail = email.trim();
      await register({ username: username.trim(), email: targetEmail, password });
      // После регистрации переходим к подтверждению адреса почты
      navigate('/verify-email', { state: { email: targetEmail } });
    } catch (err) {
      setError(err.message || 'Не удалось зарегистрироваться');
    }
  };

  return (
    <form onSubmit={handleSubmit}>
      {error && <div className="auth-error">{error}</div>}
      <input value={username} onChange={e => setUsername(e.target.value)} placeholder="Логин" />
      <input type="email" value={email} onChange={e => setEmail(e.target.value)} placeholder="Email" />
      <input type="password" value={password}
             onChange={e => setPassword(e.target.value)} placeholder="Пароль" />
      <input type="password" value={password2}
             onChange={e => setPassword2(e.target.value)} placeholder="Повторите пароль" />
      <button type="submit" className="auth-submit">Зарегистрироваться</button>
    </form>
  );
}
